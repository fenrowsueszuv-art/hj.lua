--// Utility
local function Create(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        inst[k] = v
    end
    for _, child in pairs(children or {}) do
        child.Parent = inst
    end
    return inst
end

local function Tween(obj, info, goal)
    local ts = game:GetService("TweenService")
    local t = ts:Create(obj, info, goal)
    t:Play()
    return t
end

local function Corner(radius, parent)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius)
    c.Parent = parent
    return c
end

local function Stroke(color, thickness, parent)
    local s = Instance.new("UIStroke")
    s.Color = color
    s.Thickness = thickness
    s.Parent = parent
    return s
end

local function Padding(top, right, bottom, left, parent)
    local p = Instance.new("UIPadding")
    p.PaddingTop = UDim.new(0, top)
    p.PaddingRight = UDim.new(0, right)
    p.PaddingBottom = UDim.new(0, bottom)
    p.PaddingLeft = UDim.new(0, left)
    p.Parent = parent
    return p
end

-- Populate friends
pcall(function()
    local friends = Players:GetFriendsAsync(LocalPlayer.UserId)
    local function readPage(page)
        for _, f in pairs(page:GetCurrentPage()) do
            friendsData.total += 1
            for _, p in pairs(Players:GetPlayers()) do
                if p.UserId == f.Id then
                    friendsData.inServer += 1
                end
            end
        end
        if not page.IsFinished then
            page:AdvanceToNextPageAsync()
            readPage(page)
        end
    end
    task.spawn(readPage, friends)
end)

        Size = UDim2.fromOffset(36, 36),
        BackgroundColor3 = tab.id == activeTab and Theme.Accent or Color3.fromRGB(0,0,0),
        BackgroundTransparency = tab.id == activeTab and 0 or 1,
        Text = tab.icon,
        TextColor3 = tab.id == activeTab and Color3.new(1,1,1) or Theme.TextMuted,
        TextSize = 16,
        Font = Enum.Font.GothamBold,
        ZIndex = 11,
        LayoutOrder = isProfile and 99 or i,
        AutoButtonColor = false,
    })
    btn.Parent = SideList
    Corner(8, btn)
    
    navButtons[tab.id] = btn
    
    btn.MouseEnter:Connect(function()
        if tab.id ~= activeTab then
            Tween(btn, TweenInfo.new(0.15), {BackgroundTransparency = 0.85, TextColor3 = Theme.TextPrimary})
            btn.BackgroundColor3 = Theme.Accent
        end
    end)
    btn.MouseLeave:Connect(function()
        if tab.id ~= activeTab then
            Tween(btn, TweenInfo.new(0.15), {BackgroundTransparency = 1, TextColor3 = Theme.TextMuted})
        end
    end)
    btn.MouseButton1Click:Connect(function()
        local prev = activeTab
        activeTab = tab.id
        if navButtons[prev] then
            Tween(navButtons[prev], TweenInfo.new(0.15), {BackgroundTransparency = 1, TextColor3 = Theme.TextMuted})
        end
        Tween(btn, TweenInfo.new(0.15), {BackgroundTransparency = 0, TextColor3 = Color3.new(1,1,1)})
        btn.BackgroundColor3 = Theme.Accent
    end)
end

-- Uptime tracker
local startTime = tick()
RunService.Heartbeat:Connect(function()
    local elapsed = math.floor(tick() - startTime)
    local h = math.floor(elapsed/3600)
    local m = math.floor((elapsed%3600)/60)
    local s = elapsed%60
    uptimeVal.Text = string.format("%02d:%02d:%02d", h, m, s)
    -- Update player count
    playersVal.Text = #Players:GetPlayers() .. " playing"
    -- Update ping
    pcall(function()
        pingCell.Text = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()) .. "ms"
    end)
end)

--// Dragging
local dragging, dragStart, startPos = false, nil, nil

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

--// Toggle keybind (RightControl)
local visible = true
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if input.KeyCode == Enum.KeyCode.RightControl then
        visible = not visible
        Tween(Main, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
            GroupTransparency = visible and 0 or 1
        })
        Main.Visible = visible
    end
end)
