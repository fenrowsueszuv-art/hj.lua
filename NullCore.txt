-- Slider — LayoutOrder-based
local function Slider(parent, text, flag, min, max, fmt)
    local wrap=N("Frame",{Size=UDim2.new(1,0,0,44),BackgroundColor3=Th.BG,BorderSizePixel=0,ZIndex=10})
    wrap.Parent=parent corner(7,wrap) pad(6,10,6,10,wrap)

    local valLbl=lbl({Size=UDim2.new(0,50,0,14),Position=UDim2.new(1,-50,0,0),
        BackgroundTransparency=1,Text=tostring(Flags[flag]),TextColor3=Th.Accent,
        TextSize=10,Font=Enum.Font.GothamBold,TextXAlignment=Enum.TextXAlignment.Right,ZIndex=11})
    valLbl.Parent=wrap

    lbl({Size=UDim2.new(1,-55,0,14),BackgroundTransparency=1,Text=text,TextColor3=Th.T1,
        TextSize=10,TextXAlignment=Enum.TextXAlignment.Left,ZIndex=11}).Parent=wrap

    local track=N("Frame",{Size=UDim2.new(1,0,0,5),Position=UDim2.new(0,0,1,-10),
        BackgroundColor3=Th.Border,BorderSizePixel=0,ZIndex=11}) track.Parent=wrap corner(3,track)

    local fill=N("Frame",{Size=UDim2.new((Flags[flag]-min)/(max-min),0,1,0),
        BackgroundColor3=Th.Accent,BorderSizePixel=0,ZIndex=12}) fill.Parent=track corner(3,fill)

    local dragging=false
    local function update(absX)
        local rel=math.clamp((absX-track.AbsolutePosition.X)/math.max(track.AbsoluteSize.X,1),0,1)
        local val=math.floor(min+rel*(max-min))
        Flags[flag]=val fill.Size=UDim2.new(rel,0,1,0)
        valLbl.Text=fmt and string.format(fmt,val) or tostring(val)
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=true update(i.Position.X)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType==Enum.UserInputType.MouseMovement or i.UserInputType==Enum.UserInputType.Touch) then
            update(i.Position.X)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType==Enum.UserInputType.MouseButton1 or i.UserInputType==Enum.UserInputType.Touch then
            dragging=false
        end
    end)
end

local ESPObjects={}
local function getChar(p) return p and p.Character end
local function getHum(p) local c=getChar(p) return c and c:FindFirstChildOfClass("Humanoid") end

local function makeESP(player)
    if player==LP then return end
    local obj={
        box=Drawing.new("Square"), name=Drawing.new("Text"),
        health=Drawing.new("Line"), tracer=Drawing.new("Line"),
    }
    obj.box.Visible=false obj.box.Thickness=1 obj.box.Filled=false
    obj.name.Visible=false obj.name.Size=13 obj.name.Center=true obj.name.Outline=true
    obj.name.Color=Color3.new(1,1,1)
    obj.health.Visible=false obj.health.Color=Th.Green obj.health.Thickness=2
    obj.tracer.Visible=false obj.tracer.Color=Th.Red obj.tracer.Thickness=1
    ESPObjects[player]=obj
end
local function removeESP(p)
    if ESPObjects[p] then for _,d in pairs(ESPObjects[p]) do pcall(function() d:Remove() end) end ESPObjects[p]=nil end
end
for _,p in pairs(Players:GetPlayers()) do makeESP(p) end
Players.PlayerAdded:Connect(makeESP) Players.PlayerRemoving:Connect(removeESP)

local fovCircle=Drawing.new("Circle")
fovCircle.Visible=false fovCircle.Color=Color3.new(1,1,1) fovCircle.Thickness=1 fovCircle.Filled=false fovCircle.NumSides=64

local function getClosest()
    local best,bestDist,bestChar=nil,math.huge,nil
    local center=Vector2.new(Cam.ViewportSize.X/2,Cam.ViewportSize.Y/2)
    for _,p in pairs(Players:GetPlayers()) do
        if p==LP then continue end
        local c=getChar(p) if not c then continue end
        local root=c:FindFirstChild("HumanoidRootPart") if not root then continue end
        local hum=getHum(p) if not hum or hum.Health<=0 then continue end
        local pos3,vis=Cam:WorldToViewportPoint(root.Position)
        if vis then
            local dist=(Vector2.new(pos3.X,pos3.Y)-center).Magnitude
            if dist<Flags.FovRadius and dist<bestDist then
                best=p bestDist=dist bestChar=c
            end
        end
    end
    return best,bestChar
end

local activeTracers={}
local function spawnTracer(from,to)
    local line=Drawing.new("Line") line.From=from line.To=to
    line.Color=Color3.fromRGB(255,80,80) line.Thickness=1.5 line.Visible=true
    table.insert(activeTracers,{line=line,born=tick()})
end

local hookedNC=false
local function hookNC()
    if hookedNC then return end hookedNC=true
    pcall(function()
        local mt=getrawmetatable(game) local old=mt.__namecall
        setreadonly(mt,false)
        mt.__namecall=newcclosure(function(self,...)
            local name=getnamecallmethod()
            if Flags.AntiKick and name=="Kick" and self==LP then return end
            if Flags.NukeAC and (name=="FireServer" or name=="InvokeServer") then
                local n=(self.Name or ""):lower()
                for _,k in ipairs({"exploit","cheat","detect","ban","flag","validate","integrity","security","anticheat","ac_"}) do
                    if n:find(k) then return end
                end
            end
            return old(self,...)
        end)
        setreadonly(mt,true)
    end)
end

local bsConn
local function startBS()
    if bsConn then bsConn:Disconnect() end
    bsConn=RunService.Heartbeat:Connect(function()
        if not Flags.Backshoot then bsConn:Disconnect() return end
        local _,tc=getClosest() if not tc then return end
        local tr=tc:FindFirstChild("HumanoidRootPart")
        local myRoot=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if tr and myRoot then
            myRoot.CFrame=CFrame.new(tr.Position+(-tr.CFrame.LookVector*5)+Vector3.new(0,2,0),tr.Position)
        end
    end)
end

local startTime=tick()
RunService.RenderStepped:Connect(function()
    local e=math.floor(tick()-startTime)
    uptimeL.Text=string.format("%02d:%02d:%02d",math.floor(e/3600),math.floor((e%3600)/60),e%60)
    playerL.Text=#Players:GetPlayers().." playing"
    pcall(function() pingL.Text=math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue()).."ms" end)

    if Flags.AntiKick or Flags.NukeAC then hookNC() end
    if Flags.Backshoot and (not bsConn or not bsConn.Connected) then startBS() end

    fovCircle.Visible=Flags.AimbotEnabled or Flags.SilentAim
    fovCircle.Radius=Flags.FovRadius
    fovCircle.Position=Vector2.new(Cam.ViewportSize.X/2,Cam.ViewportSize.Y/2)

    if Flags.AimbotEnabled then
        local _,tc=getClosest()
        if tc then
            local part=tc:FindFirstChild(Flags.HitPart) or tc:FindFirstChild("HumanoidRootPart")
            if part then
                Cam.CFrame=Cam.CFrame:Lerp(CFrame.new(Cam.CFrame.Position,part.Position),Flags.AimbotSmooth)
            end
        end
    end

    for i=#activeTracers,1,-1 do
        local t=activeTracers[i]
        if tick()-t.born>0.12 then t.line:Remove() table.remove(activeTracers,i) end
    end

    if Flags.Tracers then
        local _,tc=getClosest()
        if tc then
            local part=tc:FindFirstChild("HumanoidRootPart")
            local myRoot=LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
            if part and myRoot then
                local f2,v1=Cam:WorldToViewportPoint(myRoot.Position)
                local t2,v2=Cam:WorldToViewportPoint(part.Position)
                if v1 and v2 then spawnTracer(Vector2.new(f2.X,f2.Y),Vector2.new(t2.X,t2.Y)) end
            end
        end
    end

    for player,obj in pairs(ESPObjects) do
        local c=getChar(player)
        local root=c and c:FindFirstChild("HumanoidRootPart")
        local hum=getHum(player)
        if Flags.ESP and root and hum and hum.Health>0 then
            local p3,onScreen=Cam:WorldToViewportPoint(root.Position)
            if onScreen then
                local sf=1/p3.Z
                local bH=math.clamp(600*sf,20,200) local bW=bH*0.6
                local bX=p3.X-bW/2 local bY=p3.Y-bH/2
                if Flags.ESPBoxes then
                    obj.box.Visible=true obj.box.Position=Vector2.new(bX,bY) obj.box.Size=Vector2.new(bW,bH)
                    obj.box.Color=Color3.fromHSV(hum.Health/hum.MaxHealth*0.33,1,1)
                else obj.box.Visible=false end
                if Flags.ESPNames then
                    obj.name.Visible=true obj.name.Position=Vector2.new(p3.X,bY-16) obj.name.Text=player.Name
                else obj.name.Visible=false end
                if Flags.ESPHealth then
                    local hp=hum.Health/hum.MaxHealth
                    obj.health.Visible=true
                    obj.health.From=Vector2.new(bX-6,bY+bH) obj.health.To=Vector2.new(bX-6,bY+bH-(bH*hp))
                    obj.health.Color=Color3.fromHSV(hp*0.33,1,1)
                else obj.health.Visible=false end
                if Flags.ESPTracers then
                    obj.tracer.Visible=true
                    obj.tracer.From=Vector2.new(Cam.ViewportSize.X/2,Cam.ViewportSize.Y)
                    obj.tracer.To=Vector2.new(p3.X,p3.Y)
                else obj.tracer.Visible=false end
            else obj.box.Visible=false obj.name.Visible=false obj.health.Visible=false obj.tracer.Visible=false end
        else obj.box.Visible=false obj.name.Visible=false obj.health.Visible=false obj.tracer.Visible=false end
    end
end)
